#Running Python Project using Docker
mkdir my-python-project
cd my-python-project
nano app.py
nano Dockerfile

--------------------------------------------
FROM python:3.9

WORKDIR /app

COPY . .

CMD ["python3", "app.py"]
-----------------------------------------------

docker build -t my-python-project .
docker images
docker run my-python-project 
docker ps -a
docker stop

#Docker compose and Docker Swarm
mkdir docker-compose-lab
cd 
nano docker-compose.yml

---------------------------------------
version: "3.9"
services:
  web:
    image: nginx:latest
    ports:
      - "8081:80"
    volumes:
      - ./html:/usr/share/nginx/html:ro
------------------------------------------
docker-compose up -d
localhost:8081

#scale the services
nano docker-compose.yml

------------------------------------------
version: '3'
services:
  web:
    image: nginx:latest
    expose:
      - "80"
--------------------------------------------
docker compose up -d --scale web=3
